# Vendored zlib puff, kept byte-for-byte; its narrowing conversions are intentional.
set_source_files_properties(src/third_party/puff/puff.c PROPERTIES COMPILE_OPTIONS /wd4244)
# Startup-only discovery and archive building: optimized for size. The per-call paths (chunk and
# file lookups, hooks) stay in their own files at full speed.
set_source_files_properties(src/mods/mod_index.cpp src/mods/asset_mods.cpp PROPERTIES COMPILE_OPTIONS /O1)

add_library(LiteralCrab SHARED ${DI1_COMMON} ${SCRIPT_MODS} ${ASSET_MODS} ${TOYBOXES} src/di1/engine.cpp src/di1/mods.cpp src/di1/adapters.cpp src/di1/asset_loader.cpp src/di1/camera.cpp src/di1/character_freedom.cpp src/di1/character_particles.cpp src/di1/controller.cpp src/di1/gui_screens.cpp src/di1/intro.cpp src/di1/post_effects.cpp src/di1/toybox_share.cpp src/di1/main.cpp di1/exports.def)
target_sources(LiteralCrab PRIVATE ${LANGUAGE_SWITCHER})
target_sources(LiteralCrab PRIVATE src/di1/audio_banks.cpp)
target_sources(LiteralCrab PRIVATE src/di1/character_targets.cpp)
target_sources(LiteralCrab PRIVATE src/di1/diagnostics.cpp)
target_sources(LiteralCrab PRIVATE src/di1/auto_capture.cpp)

target_include_directories(LiteralCrab PRIVATE include src)
target_compile_definitions(LiteralCrab PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS LITERALCRAB_VERSION="${PROJECT_VERSION}")
target_compile_options(LiteralCrab PRIVATE /W4 /utf-8 /permissive-)
target_link_options(LiteralCrab PRIVATE "/MAP:$<TARGET_FILE_DIR:LiteralCrab>/LiteralCrab.map")
target_link_libraries(LiteralCrab PRIVATE user32 gdi32 winhttp)
add_custom_command(TARGET LiteralCrab POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_SOURCE_DIR}/di1/LiteralCrab.ini" "$<TARGET_FILE_DIR:LiteralCrab>/LiteralCrab.ini")
